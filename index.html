function saveStartDate() { const input = document.getElementById("startDateInput"); const selectedDate = input.value; if (!selectedDate) { alert("Vui lòng chọn ngày bắt đầu."); return; } const today = getToday(); if (selectedDate > today) { alert("Ngày bắt đầu không thể ở tương lai."); return; } data.startDate = selectedDate; data.date = today; data.accumulatedDays = calculateDaysFromStart( selectedDate, today ); // Lưu ngay vào máy localStorage.setItem( STORAGE_KEY, JSON.stringify(data) ); // Ghi trực tiếp lên Firebase cloudDataRef .update({ startDate: data.startDate, date: data.date, accumulatedDays: data.accumulatedDays }) .then(function() { console.log( "🔥 Đã lưu ngày bắt đầu lên Firebase:", data.startDate ); document.getElementById( "status" ).textContent = "☁️ Đã đồng bộ ngày bắt đầu."; }) .catch(function(error) { console.error( "❌ Firebase lưu ngày lỗi:", error ); document.getElementById( "status" ).textContent = "⚠️ Không thể lưu ngày lên Firebase."; }); updateDisplay(); }
